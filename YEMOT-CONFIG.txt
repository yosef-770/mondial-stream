# ============================================================
# הגדרות ימות המשיח — שלוחות SIP ל-FreeSWITCH
# ============================================================
# IP שרת (FreeSWITCH):
#   91.98.89.80:5060
#
# אל תשתמש ב-music_on_hold / זרם בימות!
#
# חזרה עם #: FreeSWITCH מנתק על # → ימות חוזר שלב אחורה
# (ברירת מחדל בימות — אל תגדיר routing_ip_end_goto=hangup)
#
# שידור כבוי: משמיעים freeswitch/sounds/he/live/off.wav
# התקנה: ./scripts/set-off-air-audio.sh /path/to/your.mp3
#
# מקורות השמע מוגדרים ב-.env:
#   STREAM_1_URL … STREAM_20_URL
# ריק = השלוחה לא משדרת.

# איכות: routing_extension=N למנוי, 100+N ללא-מנוי (3 → 103, 12 → 112).
# אותו IP ופורט. רק המספר משתנה.

# --- שלוחה עם routing_extension=1 (או kanbet) — STREAM_1_URL ---
type=routing_ip
routing_ip=91.98.89.80
routing_ip_port=5060
routing_extension=1
routing_ip_dial_time=60
